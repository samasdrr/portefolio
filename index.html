function scaleToFit() { if (window.innerWidth <= 640) return; var el = document.getElementById("cw"); el.style.transform = "none"; requestAnimationFrame(function() { var rect = el.getBoundingClientRect(); var sh = window.innerHeight * 0.96; var sw = window.innerWidth * 0.96; var s = Math.min(1, sh / rect.height, sw / rect.width); if (s < 1) { el.style.transform = "scale(" + s + ")"; el.style.transformOrigin = "top center"; document.body.style.paddingTop = Math.max(8, (window.innerHeight - rect.height * s) / 2) + "px"; } }); } window.addEventListener("load", function() { setTimeout(scaleToFit, 50); }); window.addEventListener("resize", scaleToFit);
